cmake_minimum_required(VERSION 3.28)
project(hhc-obs-plugin VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only Windows x64 is supported")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/W4 /utf-8 /permissive-)
include(CTest)
add_library(hhc-core STATIC src/capture-policy.cpp)
target_include_directories(hhc-core PUBLIC src)
add_executable(capture-test tests/capture-test.cpp)
target_link_libraries(capture-test PRIVATE hhc-core)
add_test(NAME capture-policy COMMAND capture-test)
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.deps/sdk" "${CMAKE_SOURCE_DIR}/.deps/qt" "${CMAKE_SOURCE_DIR}/.deps/prebuilt")
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
add_library(hhc-capture STATIC src/capture-output.cpp src/hls-muxer.cpp)
target_include_directories(hhc-capture PUBLIC src PRIVATE .deps/prebuilt/include)
target_link_libraries(hhc-capture PUBLIC hhc-core OBS::libobs Qt6::Core PRIVATE
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avformat.lib"
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avcodec.lib"
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avutil.lib")
add_library(hhc-obs-plugin MODULE src/plugin.cpp)
target_link_libraries(hhc-obs-plugin PRIVATE hhc-capture OBS::obs-frontend-api Qt6::Widgets)
add_executable(obs-capture-test tests/obs-capture-test.cpp)
target_link_libraries(obs-capture-test PRIVATE hhc-capture Qt6::Gui)

add_library(hhc-obs-fixture MODULE src/plugin.cpp tests/fixture-controller.cpp)
target_compile_definitions(hhc-obs-fixture PRIVATE HHC_FIXTURE_BUILD)
target_link_libraries(hhc-obs-fixture PRIVATE hhc-capture OBS::obs-frontend-api Qt6::Widgets)
