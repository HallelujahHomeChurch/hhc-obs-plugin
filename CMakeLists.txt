cmake_minimum_required(VERSION 3.28)
project(hhc-obs-plugin VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only Windows x64 is supported")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/W4 /utf-8 /permissive-)
include(CTest)
add_library(hhc-core STATIC src/capture-policy.cpp)
target_include_directories(hhc-core PUBLIC src)
add_executable(capture-test tests/capture-test.cpp)
target_link_libraries(capture-test PRIVATE hhc-core)
add_test(NAME capture-policy COMMAND capture-test)
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.deps/sdk" "${CMAKE_SOURCE_DIR}/.deps/qt" "${CMAKE_SOURCE_DIR}/.deps/prebuilt")
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Concurrent)
add_library(hhc-capture STATIC src/capture-output.cpp src/hls-muxer.cpp)
target_include_directories(hhc-capture PUBLIC src PRIVATE .deps/prebuilt/include)
target_link_libraries(hhc-capture PUBLIC hhc-core OBS::libobs Qt6::Core PRIVATE
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avformat.lib"
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avcodec.lib"
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avutil.lib")
add_library(hhc-obs-plugin MODULE src/plugin.cpp)
target_link_libraries(hhc-obs-plugin PRIVATE hhc-capture OBS::obs-frontend-api Qt6::Widgets)
add_executable(obs-capture-test tests/obs-capture-test.cpp)
target_link_libraries(obs-capture-test PRIVATE hhc-capture Qt6::Gui)

add_library(hhc-obs-fixture MODULE src/plugin.cpp tests/fixture-controller.cpp)
target_compile_definitions(hhc-obs-fixture PRIVATE HHC_FIXTURE_BUILD)
target_link_libraries(hhc-obs-fixture PRIVATE hhc-capture OBS::obs-frontend-api Qt6::Widgets)
add_library(hhc-store STATIC src/session-store.cpp)
target_link_libraries(hhc-store PUBLIC Qt6::Core)
target_include_directories(hhc-store PUBLIC src)
target_link_libraries(hhc-capture PUBLIC hhc-store)
# Fault injection is compiled into a separate developer harness, never the plugin DLL.
add_library(hhc-capture-faults STATIC src/capture-output.cpp src/hls-muxer.cpp)
target_compile_definitions(hhc-capture-faults PRIVATE HHC_CAPTURE_TEST_FAULTS)
target_include_directories(hhc-capture-faults PUBLIC src PRIVATE .deps/prebuilt/include)
target_link_libraries(hhc-capture-faults PUBLIC hhc-core hhc-store OBS::libobs Qt6::Core PRIVATE
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avformat.lib"
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avcodec.lib"
 "${CMAKE_SOURCE_DIR}/.deps/prebuilt/lib/avutil.lib")
add_executable(obs-capture-fault-test tests/obs-capture-test.cpp)
target_link_libraries(obs-capture-fault-test PRIVATE hhc-capture-faults Qt6::Gui)
add_executable(upload-test tests/upload-test.cpp)
target_link_libraries(upload-test PRIVATE hhc-store)
add_test(NAME local-journal COMMAND upload-test)
set_tests_properties(local-journal PROPERTIES ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")
add_library(hhc-auth STATIC src/auth.cpp)
target_link_libraries(hhc-auth PUBLIC Qt6::Core PRIVATE bcrypt advapi32)
target_include_directories(hhc-auth PUBLIC src)
add_library(hhc-wire STATIC src/wire.cpp)
target_include_directories(hhc-wire PUBLIC src)
target_link_libraries(hhc-wire PUBLIC Qt6::Core)
add_executable(wire-test tests/wire-test.cpp)
target_link_libraries(wire-test PRIVATE hhc-wire hhc-auth)
add_test(NAME c1-wire COMMAND wire-test)
set_tests_properties(c1-wire PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")
add_executable(auth-test tests/auth-test.cpp)
target_link_libraries(auth-test PRIVATE hhc-auth)
add_test(NAME local-auth COMMAND auth-test)
set_tests_properties(local-auth PROPERTIES ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")
add_library(hhc-dock STATIC src/dock.cpp)
target_link_libraries(hhc-dock PUBLIC Qt6::Widgets)
target_include_directories(hhc-dock PUBLIC src)
add_library(hhc-local-controller STATIC src/local-controller.cpp)
target_link_libraries(hhc-local-controller PUBLIC hhc-dock hhc-capture Qt6::Concurrent)
target_link_libraries(hhc-obs-plugin PRIVATE hhc-local-controller)
target_link_libraries(hhc-obs-fixture PRIVATE hhc-local-controller Qt6::Network)
add_executable(local-controller-test tests/local-controller-test.cpp)
target_link_libraries(local-controller-test PRIVATE hhc-local-controller)
add_executable(dock-test tests/dock-test.cpp)
target_link_libraries(dock-test PRIVATE hhc-dock)
add_test(NAME mock-dock COMMAND dock-test)
set_tests_properties(mock-dock PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH};QT_QPA_PLATFORM=windows;QT_PLUGIN_PATH=${CMAKE_SOURCE_DIR}/.deps/qt/plugins")

add_library(hhc-http STATIC src/http.cpp)
target_include_directories(hhc-http PUBLIC src)
target_link_libraries(hhc-http PUBLIC hhc-wire Qt6::Network)
add_executable(http-test tests/http-test.cpp)
target_link_libraries(http-test PRIVATE hhc-http)
add_test(NAME c1-http COMMAND http-test)
set_tests_properties(c1-http PROPERTIES ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")

add_library(hhc-native-auth STATIC src/native-auth.cpp)
target_include_directories(hhc-native-auth PUBLIC src)
target_link_libraries(hhc-native-auth PUBLIC hhc-auth hhc-http Qt6::Widgets Qt6::Concurrent)
add_executable(native-auth-test tests/native-auth-test.cpp)
target_link_libraries(native-auth-test PRIVATE hhc-native-auth)
add_test(NAME native-oauth COMMAND native-auth-test)
set_tests_properties(native-oauth PROPERTIES ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")

target_link_libraries(hhc-capture PUBLIC hhc-wire)
target_link_libraries(hhc-capture-faults PUBLIC hhc-wire)

add_library(hhc-sync STATIC src/capture-sync.cpp)
target_include_directories(hhc-sync PUBLIC src)
target_link_libraries(hhc-sync PUBLIC hhc-http hhc-store)
add_executable(sync-test tests/sync-test.cpp)
target_link_libraries(sync-test PRIVATE hhc-sync)
add_test(NAME capture-http-recovery COMMAND sync-test)
set_tests_properties(capture-http-recovery PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")

target_link_libraries(auth-preview PRIVATE hhc-native-auth)

add_library(hhc-platform-controller STATIC src/platform-controller.cpp)
target_include_directories(hhc-platform-controller PUBLIC src)
target_link_libraries(hhc-platform-controller PUBLIC hhc-dock hhc-capture hhc-native-auth hhc-sync Qt6::Concurrent)

target_link_libraries(hhc-upload-helper PRIVATE hhc-native-auth hhc-sync)

target_link_libraries(production-route-test PRIVATE hhc-http)

set_tests_properties(capture-http-finalization PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")

set_tests_properties(worker-error-propagation PROPERTIES ENVIRONMENT "PATH=${CMAKE_SOURCE_DIR}/.deps/qt/bin\;$ENV{PATH}")

target_link_libraries(production-status-test PRIVATE hhc-native-auth)

execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE HHC_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)

 target_compile_definitions(${target} PRIVATE HHC_SOURCE_COMMIT="${HHC_SOURCE_COMMIT}")
